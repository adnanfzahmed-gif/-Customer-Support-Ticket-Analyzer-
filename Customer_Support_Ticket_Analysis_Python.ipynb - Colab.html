def clean_text(text):

    #  Convert text to lowercase
    text = text.lower()

    #  Remove punctuation
    for symbol in ['.', ',', '!', '?', '-']:
        text = text.replace(symbol, ' ')

    #  Replace shorthand "ok" with "okay"
    # Add spaces around the word so "okay" is not changed accidentally.
    text = ' '.join(text.split())
    text = text.replace(' ok ', ' okay ')

    #  Normalize multiple spaces
    text = ' '.join(text.split())

    #  Remove leading/trailing spaces
    text = text.strip()

    return text


#  Clean all issue descriptions
for i in range(len(ticket_data['Issue_Description'])):

    ticket_data['Issue_Description'][i] = clean_text(
        ticket_data['Issue_Description'][i]
    )


print(" All issue descriptions have been cleaned successfully!")

def count_tickets_with_word(word):

    count = 0

    #  Convert search word to lowercase
    word = word.lower().strip()

    for description in ticket_data['Issue_Description']:

        #  Convert description into individual words
        words = description.lower().split()

        #  Check whether the word exists
        if word in words:
            count += 1

    return count


#  Required keywords

Overall, this project demonstrates how Python data structures, loops, functions, string methods, sets, and conditional statements can be used to organise customer support data and generate meaningful business insights.
